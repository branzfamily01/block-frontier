  function key(x, y, z) { return `${x},${y},${z}`; }
  function getBlock(x, y, z) { return world.get(key(x, y, z)) || 0; }
  function setBlock(x, y, z, type) {
    const k = key(x, y, z);
    if (type) world.set(k, type); else world.delete(k);
    worldDirty = true;
    scheduleSave();

  function hash2(x, z) {
    const s = Math.sin(x * 127.1 + z * 311.7) * 43758.5453123;
    return s - Math.floor(s);
  }

  function terrainHeight(x, z) {
    const nx = x - WORLD_SIZE / 2;
    const nz = z - WORLD_SIZE / 2;
    const wave = Math.sin(nx * 0.31) * 0.7 + Math.cos(nz * 0.27) * 0.6 + Math.sin((nx + nz) * 0.16) * 0.45;
    let h = 2;
    if (wave > 0.28) h += 1;
    if (wave > 1.05) h += 1;
    return h;
  }

  function generateWorld() {
    world.clear();
    for (let x = 0; x < WORLD_SIZE; x++) {
      for (let z = 0; z < WORLD_SIZE; z++) {
        const h = terrainHeight(x, z);
        world.set(key(x, 0, z), 8);
        for (let y = 1; y < h; y++) {
          world.set(key(x, y, z), y <= 1 ? 3 : 2);
        }
        const edge = x < 2 || z < 2 || x > WORLD_SIZE - 3 || z > WORLD_SIZE - 3;
        world.set(key(x, h, z), edge ? 4 : 1);

        const dx = x - WORLD_SIZE / 2;
        const dz = z - WORLD_SIZE / 2;
        const awayFromSpawn = Math.hypot(dx, dz) > 5;
        const r = hash2(x, z);
        if (!edge && awayFromSpawn && r > 0.945) addTree(x, h + 1, z);
        if (!edge && awayFromSpawn && r > 0.905 && r < 0.918) {
          world.set(key(x, h + 1, z), 6);
        }
      }
    }
    worldDirty = true;
    spawnPlayer(true);
  }

  function addTree(x, y, z) {
    const trunk = 2 + Math.floor(hash2(x + 9, z + 3) * 2);
    for (let i = 0; i < trunk; i++) world.set(key(x, y + i, z), 5);
    const top = y + trunk;
    for (let ox = -1; ox <= 1; ox++) {
      for (let oz = -1; oz <= 1; oz++) {
        if (Math.abs(ox) + Math.abs(oz) <= 2) world.set(key(x + ox, top, z + oz), 7);
      }
    }
    world.set(key(x, top + 1, z), 7);
  }

  function highestBlockAt(x, z) {
    for (let y = MAX_BUILD_Y; y >= 0; y--) if (getBlock(x, y, z)) return y;
    return 0;
  }

  function spawnPlayer(force = false) {
    if (!force && Number.isFinite(player.y)) return;
    const sx = Math.floor(WORLD_SIZE / 2);
    const sz = Math.floor(WORLD_SIZE / 2);
    player.x = sx + 0.5;
    player.z = sz + 0.5;
    player.y = highestBlockAt(sx, sz) + 1.01;
    player.vy = 0;
    player.grounded = false;
  }

  function saveNow() {
    try {
      localStorage.setItem(SAVE_KEY, JSON.stringify({
        v: 1,
        world: Array.from(world.entries()),
        selectedType,
        player: { x: player.x, y: player.y, z: player.z, yaw: player.yaw, pitch: player.pitch },
        soundEnabled,
      }));
    } catch (_) {}
  }

  function scheduleSave() {
    clearTimeout(saveTimer);
    saveTimer = setTimeout(saveNow, 450);
